#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#ifdef _WIN32
    #include <windows.h>
    #define ROOT_DIR "C:\\*"
    #define PATH_MAX_LEN MAX_PATH
#else
    #define _XOPEN_SOURCE 500
    #include <ftw.h>
    #include <limits.h>
    #define ROOT_DIR "/"
    #define PATH_MAX_LEN PATH_MAX
#endif

// Global variable to store the found path and target name
char found_path[PATH_MAX_LEN] = "";
const char *target_filename = "image.png";

// Helper function for case-insensitive comparison (cross-platform)
int case_insensitive_compare(const char *s1, const char *s2) {
    #ifdef _WIN32
        return _stricmp(s1, s2);
    #else
        return strcasecmp(s1, s2);
    #endif
}

#ifdef _WIN32
// --- WINDOWS IMPLEMENTATION (Recursive Function) ---
void find_image_windows(const char *search_mask) {
    WIN32_FIND_DATAA find_data;
    HANDLE hFind = FindFirstFileA(search_mask, &find_data);

    if (hFind == INVALID_HANDLE_VALUE) return; // Skip directories we can't open (PermissionError)

    // Extract base directory path from mask (e.g., "C:\* " -> "C:\")
    char base_dir[PATH_MAX_LEN];
    strcpy(base_dir, search_mask);
    char *last_slash = strrchr(base_dir, '\\');
    if (last_slash) *(last_slash + 1) = '\0';

    do {
        // Skip current directory "." and parent directory ".."
        if (strcmp(find_data.cFileName, ".") == 0 || strcmp(find_data.cFileName, "..") == 0) {
            continue;
        }

        // Construct full path
        char full_path[PATH_MAX_LEN];
        snprintf(full_path, sizeof(full_path), "%s%s", base_dir, find_data.cFileName);

        if (find_data.dwFileAttributes & FILE_ATTRIBUTE_DIRECTORY) {
            // It's a directory -> Recurse into it safely
            char next_mask[PATH_MAX_LEN];
            snprintf(next_mask, sizeof(next_mask), "%s\\*", full_path);
            find_image_windows(next_mask);
        } else {
            // It's a file -> Match name case-insensitively
            if (case_insensitive_compare(find_data.cFileName, target_filename) == 0) {
                GetFullPathNameA(full_path, PATH_MAX_LEN, found_path, NULL);
                FindClose(hFind);
                return;
            }
        }

        // Stop early if found in a recursive call
        if (strlen(found_path) > 0) {
            FindClose(hFind);
            return;
        }

    } while (FindNextFileA(hFind, &find_data));

    FindClose(hFind);
}

#else
// --- LINUX / MACOS IMPLEMENTATION (Callback function for nftw) ---
int find_image_posix(const char *fpath, const struct stat *sb, int tflag, struct FTW *ftwbuf) {
    // FTW_F means it is a regular file
    if (tflag == FTW_F) {
        // Get only the filename from the path
        const char *filename = fpath + ftwbuf->base;
        if (case_insensitive_compare(filename, target_filename) == 0) {
            realpath(fpath, found_path); // Get absolute path
            return 1; // Stop the search immediately
        }
    }
    return 0; // Keep searching (nftw skips automatically on permission errors)
}
#endif

int main() {
    #ifdef _WIN32
        printf("Searching entire PC starting from C:\\... This may take a minute.\n");
        find_image_windows(ROOT_DIR);
    #else
        printf("Searching entire PC starting from /... This may take a minute.\n");
        // nftw arguments: root path, callback function, max open file descriptors, flags
        nftw(ROOT_DIR, find_image_posix, 20, FTW_PHYS);
    #endif

    // Print results matching Python logic
    if (strlen(found_path) > 0) {
        printf("\n Success! Image found at:\n%s\n", found_path);
    } else {
        printf("\n Could not find an image named '%s' anywhere on this drive.\n", target_filename);
    }

    return 0;
}
